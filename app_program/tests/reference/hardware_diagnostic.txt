// GROVE 7:

// ============================================================

#define LEFT_ENC_B 1

// Grove 7
#define RIGHT_ENC_A 7
#define RIGHT_ENC_B 28


volatile int32_t left_encoder_count = 0;
volatile int32_t right_encoder_count = 0;


// ============================================================
// IR SENSORS
// ============================================================

// Grove 5 = LEFT line sensor
#define IR_LEFT_PIN 6

// Grove 2 = RIGHT line sensor
#define IR_RIGHT_PIN 2

// Grove 6 = barcode sensor
//
// PRIMARY = GP26
//
// GP27 is temporarily monitored as an alternate
// diagnostic input in case the Grove cable has the
// signal on the second GPIO conductor.

#define IR_BARCODE_PIN      26
#define IR_BARCODE_ALT_PIN  27


// Current RAW state

volatile int ir_left_raw = 0;
volatile int ir_right_raw = 0;

volatile int ir_barcode_raw = 0;
volatile int ir_barcode_alt_raw = 0;


// Previous RAW state

int ir_left_previous = 0;
int ir_right_previous = 0;

int ir_barcode_previous = 0;
int ir_barcode_alt_previous = 0;


// Transition counters

uint32_t ir_left_changes = 0;
uint32_t ir_right_changes = 0;

uint32_t ir_barcode_changes = 0;
uint32_t ir_barcode_alt_changes = 0;


// ============================================================

// ============================================================
// MOTOR INITIALISATION
// ============================================================

void init_motor_pwm_pin(uint pin)
{
    gpio_set_function(
        pin,
        GPIO_FUNC_PWM
    );

    uint slice =
        pwm_gpio_to_slice_num(pin);

    pwm_set_wrap(
        slice,
        255
    );

    pwm_set_enabled(
        slice,
        true
    );
}


// ============================================================
// RAW MOTOR COMMAND
// ============================================================

void set_motor(
    uint pin_a,
    uint pin_b,
    int speed
)
{
    if (speed > 255)
        speed = 255;

    if (speed < -255)
        speed = -255;


    if (speed > 0)
    {
        pwm_set_gpio_level(
            pin_a,
            speed
        );

        pwm_set_gpio_level(
            pin_b,
            0
        );
    }

    else if (speed < 0)
    {
        pwm_set_gpio_level(
            pin_a,
            0
        );

        pwm_set_gpio_level(
            pin_b,
            -speed
        );
    }

    else
    {
        pwm_set_gpio_level(
            pin_a,
            0
        );

        pwm_set_gpio_level(
            pin_b,
            0
        );
    }
}


// ============================================================
// VEHICLE DRIVE
// ============================================================
//
// IMPORTANT:
//
// Your physical test showed:
//
// +LEFT command actually drives LEFT wheel backwards.
// +RIGHT command correctly drives RIGHT wheel forwards.
//
// Therefore LEFT motor is inverted HERE.
//
// This lets the rest of the program use:
//      +,+ = forward
//      -,- = backward
//      -,+ = left
//      +,- = right
//
// ============================================================

void drive(
    int left_speed,
    int right_speed
)
{
    // LEFT MOTOR IS PHYSICALLY REVERSED
    set_motor(
        M1_A,
        M1_B,
        -left_speed
    );


    // RIGHT MOTOR
    set_motor(
        M2_A,
        M2_B,
        right_speed
    );
}


void stop_motors()
{
    drive(0, 0);
}


// ============================================================
// ENCODER CALLBACK
// ============================================================
//
// Only encoders use interrupts.
//
// IR sensors are now POLLED continuously.
//
// ============================================================

void encoder_callback(
    uint gpio,
    uint32_t events
)
{
    (void)events;


    // ========================================================
    // LEFT ENCODER
    // ========================================================

    if (gpio == LEFT_ENC_A)
    {
        bool a =
            gpio_get(LEFT_ENC_A);

        bool b =
            gpio_get(LEFT_ENC_B);


        if (a == b)
        {
            left_encoder_count++;
        }
        else
        {
            left_encoder_count--;
        }
    }


    // ========================================================
    // RIGHT ENCODER
    // ========================================================

    else if (gpio == RIGHT_ENC_A)
    {
        bool a =
            gpio_get(RIGHT_ENC_A);

        bool b =
            gpio_get(RIGHT_ENC_B);


        if (a == b)
        {
            right_encoder_count++;
        }
        else
        {
            right_encoder_count--;
        }
    }
}


// ============================================================
// ENCODER INIT
// ============================================================

void init_encoders()
{
    gpio_init(LEFT_ENC_A);
    gpio_set_dir(LEFT_ENC_A, GPIO_IN);
    gpio_pull_up(LEFT_ENC_A);


    gpio_init(LEFT_ENC_B);
    gpio_set_dir(LEFT_ENC_B, GPIO_IN);
    gpio_pull_up(LEFT_ENC_B);


    gpio_init(RIGHT_ENC_A);
    gpio_set_dir(RIGHT_ENC_A, GPIO_IN);
    gpio_pull_up(RIGHT_ENC_A);


    gpio_init(RIGHT_ENC_B);
    gpio_set_dir(RIGHT_ENC_B, GPIO_IN);
    gpio_pull_up(RIGHT_ENC_B);


    // Install callback

    gpio_set_irq_enabled_with_callback(
        LEFT_ENC_A,
        GPIO_IRQ_EDGE_RISE |
        GPIO_IRQ_EDGE_FALL,
        true,
        &encoder_callback
    );


    gpio_set_irq_enabled(
        RIGHT_ENC_A,
        GPIO_IRQ_EDGE_RISE |
        GPIO_IRQ_EDGE_FALL,
        true
    );
}


// ============================================================
// IR INITIALISATION
// ============================================================

void init_ir()
{
    // LEFT LINE

    gpio_init(IR_LEFT_PIN);

    gpio_set_dir(
        IR_LEFT_PIN,
        GPIO_IN
    );


    // RIGHT LINE

    gpio_init(IR_RIGHT_PIN);

    gpio_set_dir(
        IR_RIGHT_PIN,
        GPIO_IN
    );


    // BARCODE PRIMARY

    gpio_init(IR_BARCODE_PIN);

    gpio_set_dir(
        IR_BARCODE_PIN,
        GPIO_IN
    );


    // BARCODE ALTERNATE GROVE 6 DATA LINE

    gpio_init(IR_BARCODE_ALT_PIN);

    gpio_set_dir(
        IR_BARCODE_ALT_PIN,
        GPIO_IN
    );


    // Initial readings

    ir_left_raw =
        gpio_get(IR_LEFT_PIN);

    ir_right_raw =
        gpio_get(IR_RIGHT_PIN);

    ir_barcode_raw =
        gpio_get(IR_BARCODE_PIN);

    ir_barcode_alt_raw =
        gpio_get(IR_BARCODE_ALT_PIN);


    ir_left_previous =
        ir_left_raw;

    ir_right_previous =
        ir_right_raw;

    ir_barcode_previous =
        ir_barcode_raw;

    ir_barcode_alt_previous =
        ir_barcode_alt_raw;
}


// ============================================================
// CONTINUOUS IR SAMPLE
// ============================================================
//
// Call this VERY frequently.
//
// No printf occurs here.
//
// ============================================================

void sample_all_ir()
{
    int left =
        gpio_get(IR_LEFT_PIN);

    int right =
        gpio_get(IR_RIGHT_PIN);

    int barcode =
        gpio_get(IR_BARCODE_PIN);

    int barcode_alt =
        gpio_get(IR_BARCODE_ALT_PIN);


    // ========================================================
    // LEFT
    // ========================================================

    if (left != ir_left_previous)
    {
        ir_left_changes++;

        ir_left_previous =
            left;
    }


    // ========================================================
    // RIGHT
    // ========================================================

    if (right != ir_right_previous)
    {
        ir_right_changes++;

        ir_right_previous =
            right;
    }


    // ========================================================
    // BARCODE GP26
    // ========================================================

    if (barcode != ir_barcode_previous)
    {
        ir_barcode_changes++;

        ir_barcode_previous =
            barcode;
    }


    // ========================================================
    // BARCODE ALTERNATE GP27
    // ========================================================

    if (barcode_alt != ir_barcode_alt_previous)
    {
        ir_barcode_alt_changes++;

        ir_barcode_alt_previous =
            barcode_alt;
    }


    ir_left_raw =
        left;

    ir_right_raw =
        right;

    ir_barcode_raw =
        barcode;

    ir_barcode_alt_raw =
        barcode_alt;
}


// ============================================================
// LIVE IR STATUS
// ============================================================

void print_ir_status()
{
    printf(
        "IR | LEFT G5/GP6=%d | RIGHT G2/GP2=%d | "
        "BARCODE G6/GP26=%d | G6 ALT/GP27=%d\n",
        ir_left_raw,
        ir_right_raw,
        ir_barcode_raw,
        ir_barcode_alt_raw
    );


    printf(
        "IR transitions | LEFT=%lu RIGHT=%lu "
        "BARCODE26=%lu BARCODE27=%lu\n",
        (unsigned long)ir_left_changes,
        (unsigned long)ir_right_changes,
        (unsigned long)ir_barcode_changes,
        (unsigned long)ir_barcode_alt_changes
    );
}


// ============================================================
// POLLING WAIT
// ============================================================
//
// Replacement for normal sleep_ms() during diagnostics.
//
// IR sensors continue being sampled approximately every 1 ms.
//
// ============================================================

void wait_with_ir(
    uint32_t milliseconds,
    bool show_ir
)
{
    uint64_t start =
        time_us_64();

    uint64_t last_print =
        time_us_64();


    while (
        (time_us_64() - start)
        <
        ((uint64_t)milliseconds * 1000)
    )
    {
        // CONSTANT IR READING

        sample_all_ir();


        if (
            show_ir &&
            (
                time_us_64() -
                last_print
                >=
                100000
            )
        )
        {
            print_ir_status();

            last_print =
                time_us_64();
        }


        sleep_us(1000);
    }
}

}


// ============================================================
// ULTRASONIC READ
// ============================================================
//
// Even while waiting for ECHO,
// we continue sampling the IR sensors.
//
// ============================================================

float read_distance_cm()
{
    gpio_put(
        TRIG_PIN,
        0
    );

    sleep_us(3);


    gpio_put(
        TRIG_PIN,
        1
    );

    sleep_us(10);


    gpio_put(
        TRIG_PIN,
        0
    );


    // ========================================================
    // WAIT FOR ECHO HIGH
    // ========================================================

    uint64_t wait_start =
        time_us_64();


    while (!gpio_get(ECHO_PIN))
    {
        sample_all_ir();


        if (
            time_us_64() -
            wait_start >
            30000
        )
        {
            return -1.0f;
        }
    }


    // ========================================================
    // MEASURE HIGH PULSE
    // ========================================================

    uint64_t pulse_start =
        time_us_64();


    while (gpio_get(ECHO_PIN))
    {
        sample_all_ir();


        if (
            time_us_64() -
            pulse_start >
            30000
        )
        {
            return -1.0f;
        }
    }


    uint64_t pulse_end =
        time_us_64();


    return
        (pulse_end - pulse_start)
        / 58.0f;
}


// ============================================================
// ULTRASONIC TEST
// ============================================================

void test_ultrasonic()
{
    printf("\n");
    printf("ULTRASONIC TEST - GROVE 3\n");
    printf("----------------------------------------\n");


    for (int i = 0; i < 3; i++)
    {
        float distance =
            read_distance_cm();


        if (distance < 0)
        {
            printf(
                "Reading %d: TIMEOUT\n",
                i + 1
            );
        }

        else
        {
            printf(
                "Reading %d: %.2f cm\n",
                i + 1,
                distance
            );
        }


        wait_with_ir(
            70,
            false
        );
    }


    printf("----------------------------------------\n");
}

// ============================================================
// I2C CHECK
// ============================================================

bool i2c_device_alive(
    uint8_t address
)
{
    uint8_t dummy;


    int result =
        i2c_read_timeout_us(
            IMU_I2C,
            address,
            &dummy,
            1,
            false,
            2000
        );


    return result >= 0;
}

// ============================================================
// SERVO ANGLE
// ============================================================

void set_servo_angle(
    int angle
)
{
    // Small diagnostic range

    if (angle < 70)
        angle = 70;

    if (angle > 110)
        angle = 110;


    float pulse_us =
        500.0f +
        ((float)angle / 180.0f)
        * 2000.0f;


    uint16_t level =
        (uint16_t)(
            pulse_us *
            1.953125f
        );


    pwm_set_gpio_level(
        SERVO_PIN,
        level
    );


    printf(
        "Servo -> %d deg\n",
        angle
    );
}


// ============================================================
// MOTOR TEST
// ============================================================

void run_motion(
    const char *name,
    int left_speed,
    int right_speed
)
{
    printf("\n");
    printf("========================================\n");
    printf("MOTOR TEST: %s\n", name);
    printf("========================================\n");


    int32_t left_before =
        left_encoder_count;

    int32_t right_before =
        right_encoder_count;


    printf(
        "Vehicle command LEFT=%d RIGHT=%d\n",
        left_speed,
        right_speed
    );


    drive(
        left_speed,
        right_speed
    );


    // IR CONTINUES TO BE READ DURING MOVEMENT

    wait_with_ir(
        MOVE_TIME_MS,
        true
    );


    stop_motors();


    wait_with_ir(
        200,
        false
    );


    int32_t left_delta =
        left_encoder_count -
        left_before;


    int32_t right_delta =
        right_encoder_count -
        right_before;


    printf(
        "LEFT encoder delta : %ld\n",
        (long)left_delta
    );


    printf(
        "RIGHT encoder delta: %ld\n",
        (long)right_delta
    );


    printf("MOTORS STOPPED\n");


    print_ir_status();


    printf(
        "========================================\n"
    );


    wait_with_ir(
        1000,
        false
    );
}


// ============================================================
// BARCODE DIAGNOSTIC
// ============================================================

void barcode_diagnostic()
{
    printf("\n");
    printf("================================================\n");
    printf("BARCODE SENSOR DIAGNOSTIC - GROVE 6\n");
    printf("================================================\n");

    printf(
        "Move the BARCODE sensor across BLACK/WHITE now.\n"
    );

    printf(
        "Watching BOTH Grove 6 GPIOs:\n"
    );

    printf(
        "  GP26 = intended barcode input\n"
    );

    printf(
        "  GP27 = alternate diagnostic input\n"
    );


    uint32_t before26 =
        ir_barcode_changes;

    uint32_t before27 =
        ir_barcode_alt_changes;


    // 8 seconds

    wait_with_ir(
        8000,
        true
    );


    uint32_t change26 =
        ir_barcode_changes -
        before26;

    uint32_t change27 =
        ir_barcode_alt_changes -
        before27;


    printf("\n");
    printf("BARCODE DIAGNOSTIC RESULT\n");

    printf(
        "GP26 transitions = %lu\n",
        (unsigned long)change26
    );

    printf(
        "GP27 transitions = %lu\n",
        (unsigned long)change27
    );


    if (
        change26 > 0 &&
        change27 == 0
    )
    {
        printf(
            "*** BARCODE SIGNAL APPEARS TO BE GP26 ***\n"
        );
    }

    else if (
        change27 > 0 &&
        change26 == 0
    )
    {
        printf(
            "*** BARCODE SIGNAL APPEARS TO BE GP27 ***\n"
        );

        printf(
            "*** CHANGE IR_BARCODE_PIN TO 27 AFTER CONFIRMING ***\n"
        );
    }

    else if (
        change26 == 0 &&
        change27 == 0
    )
    {
        printf(
            "*** NO BARCODE RESPONSE ON GP26 OR GP27 ***\n"
        );

        printf(
            "*** CHECK SENSOR POWER / OUT WIRE / HEIGHT / THRESHOLD ***\n"
        );
    }

    else
    {
        printf(
            "*** BOTH PINS CHANGED - CHECK GROVE 6 WIRING / FLOATING INPUT ***\n"
        );
    }


    printf(
        "================================================\n"
    );
}
